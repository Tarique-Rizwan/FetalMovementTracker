import React, { useEffect, useState } from 'react';
import { View, Text, FlatList, StyleSheet, TouchableOpacity } from 'react-native';
import { useIsFocused } from '@react-navigation/native';
import { Ionicons } from '@expo/vector-icons'; // Using Expo icons
import { getSessions } from '../utils/storage';
import InfoModal from '../components/InfoModal';

const HomeScreen = ({ navigation }) => {
  const [sessions, setSessions] = useState([]);
  const [modalVisible, setModalVisible] = useState(false);
  const isFocused = useIsFocused(); 

  [cite_start]// Reload data whenever screen comes into focus [cite: 33]
  useEffect(() => {
    if (isFocused) {
      loadData();
    }
  }, [isFocused]);

  const loadData = async () => {
    const data = await getSessions();
    setSessions(data);
  };

  [cite_start]// [cite: 12] Show list of past records
  const renderItem = ({ item }) => {
    const dateObj = new Date(item.date);
    const dateStr = dateObj.toLocaleDateString();
    const timeStr = dateObj.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
    const durationMins = Math.floor(item.duration / 60); 
    const durationSecs = item.duration % 60;

    return (
      <View style={styles.card}>
        <View>
          <Text style={styles.cardDate}>{dateStr} • {timeStr}</Text>
          <Text style={styles.cardKicks}>{item.kicks} Movements</Text>
        </View>
        <View style={styles.durationContainer}>
          <Text style={styles.durationText}>{durationMins}m {durationSecs}s</Text>
          <Text style={styles.durationLabel}>Duration</Text>
        </View>
      </View>
    );
  };

  return (
    <View style={styles.container}>
      {/* Header */}
      <View style={styles.header}>
        <Text style={styles.headerTitle}>Fetal Tracker</Text>
        [cite_start]{/* [cite: 25, 27] Info Button */}
        <TouchableOpacity onPress={() => setModalVisible(true)}>
          <Ionicons name="information-circle-outline" size={32} color="#333" />
        </TouchableOpacity>
      </View>

      [cite_start]{/* Static Article Section [cite: 10] */}
      <View style={styles.articleSection}>
        <Text style={styles.articleTitle}>Why track movements?</Text>
        <Text style={styles.articleDesc}>
          Monitoring your baby's activity pattern is an excellent way to gauge their well-being.
        </Text>
      </View>

      <Text style={styles.sectionHeader}>History</Text>

      <FlatList
        data={sessions}
        renderItem={renderItem}
        keyExtractor={item => item.id}
        contentContainerStyle={styles.listContent}
        ListEmptyComponent={
          <Text style={styles.emptyText}>No sessions recorded yet.</Text>
        }
      />

      {/* FAB to Start Tracking */}
      <TouchableOpacity 
        style={styles.fab} 
        onPress={() => navigation.navigate('Counter')}
      >
        <Ionicons name="add" size={32} color="white" />
      </TouchableOpacity>

      <InfoModal visible={modalVisible} onClose={() => setModalVisible(false)} />
    </View>
  );
};
