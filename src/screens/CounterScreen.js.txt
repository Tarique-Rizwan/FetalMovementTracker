import React, { useState, useEffect } from 'react';
import { View, Text, StyleSheet, TouchableOpacity } from 'react-native';
import { saveSession } from '../utils/storage';

const CounterScreen = ({ navigation }) => {
  const [seconds, setSeconds] = useState(0);
  const [kicks, setKicks] = useState(0);
  const [isRunning, setIsRunning] = useState(true);

  [cite_start]// Timer Logic [cite: 18]
  useEffect(() => {
    let interval = null;
    if (isRunning) {
      interval = setInterval(() => {
        setSeconds(prev => prev + 1);
      }, 1000);
    }
    return () => clearInterval(interval);
  }, [isRunning]);

  const handleKick = () => {
    setKicks(prev => prev + 1);
  };

  const formatTime = (totalSeconds) => {
    const mins = Math.floor(totalSeconds / 60);
    const secs = totalSeconds % 60;
    return `${mins < 10 ? '0' : ''}${mins}:${secs < 10 ? '0' : ''}${secs}`;
  };

  const handleSave = async () => {
    [cite_start]// [cite: 20] Save session
    const session = {
      id: Date.now().toString(),
      date: new Date().toISOString(),
      duration: seconds,
      kicks: kicks
    };

    await saveSession(session);
    navigation.goBack(); // Return to Home
  };

  return (
    <View style={styles.container}>
      <Text style={styles.timerText}>{formatTime(seconds)}</Text>
      <Text style={styles.subText}>Duration</Text>

      {/* Main Interaction Button */}
      <TouchableOpacity style={styles.kickButton} onPress={handleKick}>
        <Text style={styles.kickButtonText}>Tap to Kick</Text>
        <Text style={styles.kickCount}>{kicks} Kicks</Text>
      </TouchableOpacity>

      <View style={styles.buttonContainer}>
        [cite_start]{/* [cite: 22] Go back without saving */}
        <TouchableOpacity 
          style={[styles.actionButton, styles.cancelBtn]} 
          onPress={() => navigation.goBack()}
        >
          <Text style={styles.btnText}>Cancel</Text>
        </TouchableOpacity>

        [cite_start]{/* [cite: 20] Save Session */}
        <TouchableOpacity 
          style={[styles.actionButton, styles.saveBtn]} 
          onPress={handleSave}
        >
          <Text style={styles.btnText}>Save Session</Text>
        </TouchableOpacity>
      </View>
    </View>
  );
};

const styles = StyleSheet.create({

  },

    shadowOffset: { width: 0, height: 10 },
    shadowOpacity: 0.3,

    borderWidth: 5,
    borderColor: '#FF6B6B',
  },
  kickButtonText: {
    fontSize: 22,

  },
